Bronze Layer

-- raw_tmdb_movies id BIGINT PRIMARY KEY tmdb_id INTEGER NOT NULL raw_payload JSONB source_url TEXT ingested_at TIMESTAMPTZ batch_id UUID -- raw_omdb_ratings imdb_id VARCHAR(12) PK raw_payload JSONB fetched_at TIMESTAMPTZ -- raw_watchmode_streams tmdb_id INTEGER raw_payload JSONB fetched_at TIMESTAMPTZ
Silver Layer
CLEANED · NORMALIZED · VALIDATED
-- stg_movies movie_id INTEGER PRIMARY KEY title VARCHAR(255) release_dateDATE genres TEXT[] budget_usd NUMERIC(15,2) revenue_usd NUMERIC(15,2) popularity FLOAT imdb_rating NUMERIC(3,1) rt_score SMALLINT mc_score SMALLINT dq_flag BOOLEAN updated_at TIMESTAMPTZ
Gold Layer
AGGREGATED · ANALYTICS-READY
-- fct_box_office_weekly week_key DATE PK movie_id INTEGER FK week_rank SMALLINT worldwide_revenue NUMERIC(15,2) budget NUMERIC(15,2) roi_pct FLOAT weeks_out SMALLINT popularity FLOAT -- dim_movies (SCD Type 2) movie_sk SERIAL PK movie_id INTEGER valid_from DATE valid_to DATE is_current BOOLEAN
Airflow DAG — box_office_pipeline
# dags/box_office_pipeline.py from airflow import DAG from airflow.operators.python import PythonOperator from airflow.providers.postgres.operators.postgres import PostgresOperator from datetime import datetime, timedelta default_args = { 'owner': 'cinedata', 'retries': 2, 'retry_delay': timedelta(minutes=5), 'email_on_failure': True, } with DAG( dag_id='box_office_pipeline', default_args=default_args, schedule_interval='0 9 * * 1', # Every Monday 09:00 start_date=datetime(2026, 1, 1), catchup=False, tags=['cinedata', 'box-office'], ) as dag: ingest_tmdb = PythonOperator( task_id='ingest_tmdb_now_playing', python_callable=fetch_tmdb_now_playing, ) ingest_omdb = PythonOperator( task_id='ingest_omdb_ratings', python_callable=fetch_omdb_batch, ) ingest_watchmode = PythonOperator( task_id='ingest_watchmode_streams', python_callable=fetch_watchmode_availability, ) transform_silver = PostgresOperator( task_id='transform_to_silver', sql='sql/silver/stg_movies.sql', ) transform_gold = PostgresOperator( task_id='transform_to_gold', sql='sql/gold/fct_box_office_weekly.sql', ) dq_checks = PythonOperator( task_id='run_data_quality_checks', python_callable=run_dq_suite, ) notify = PythonOperator( task_id='notify_pipeline_complete', python_callable=send_slack_notification, ) [ingest_tmdb, ingest_omdb, ingest_watchmode] >> transform_silver transform_silver >> transform_gold >> dq_checks >> notify
Python Ingest Script — ingest_tmdb.py
# ingestion/ingest_tmdb.py import os, requests, psycopg2, json from datetime import datetime import uuid TMDB_KEY = os.environ['TMDB_API_KEY'] DB_CONN = os.environ['POSTGRES_CONN'] def fetch_now_playing(page=1): url = f"https://api.themoviedb.org/3/movie/now_playing" params = {"api_key": TMDB_KEY, "region": "US", "page": page} r = requests.get(url, params=params, timeout=10) r.raise_for_status() return r.json()["results"] def fetch_movie_detail(tmdb_id): url = f"https://api.themoviedb.org/3/movie/{tmdb_id}" r = requests.get(url, params={"api_key": TMDB_KEY}, timeout=10) r.raise_for_status() return r.json() def load_to_bronze(conn, records): batch_id = str(uuid.uuid4()) with conn.cursor() as cur: for rec in records: cur.execute(""" INSERT INTO bronze.raw_tmdb_movies (tmdb_id, raw_payload, source_url, ingested_at, batch_id) VALUES (%s, %s, %s, %s, %s) ON CONFLICT (tmdb_id) DO UPDATE SET raw_payload = EXCLUDED.raw_payload, ingested_at = EXCLUDED.ingested_at """, ( rec["id"], json.dumps(rec), f"tmdb/movie/{rec['id']}", datetime.utcnow(), batch_id, )) conn.commit() return len(records) def run(): conn = psycopg2.connect(DB_CONN) movies = fetch_now_playing() detailed = [fetch_movie_detail(m["id"]) for m in movies] n = load_to_bronze(conn, detailed) print(f"[BRONZE] Loaded {n} records — batch complete") conn.close() if __name__ == "__main__": run()
